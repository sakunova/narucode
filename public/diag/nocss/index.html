const master = slider(0.667, 0, 1)

const _rlpfHz = (knob, knobMax=5, maxHz=5000) =>
  knob.fmap(x => (x / knobMax) * maxHz)

Pattern.prototype.rlpf = function(knob){
  return this.lpf(_rlpfHz(knob, 5, 5000))
}

window.ar = function(...input) {

  if (input.length % 2 !== 0) {
    throw new Error('Arrange needs a length paramter for each pattern (length, pattern, length, pattern)');
  }
  let sects = [];
  let total = 0;
  for (let i = 0; i < input.length; i += 2) {
    let inp = [input.at(i)].flat()
    let cycles = inp.at(0);
    let start = inp.at(1) ?? 0
    total += cycles;

    let pattern = input.at(i + 1);
    sects.push([cycles, pattern.ribbon(start, cycles).fast(cycles)]);
  }
  return stepcat(...sects).slow(total);
}


window.p = stack

window.blockArrange = function (patArr, modifiers = []) {
  return stack(
    ...patArr.map(([pat, maskPat]) => {
      pat = [pat].flat()

      return maskPat.fmap(m => {
        return stack(...pat.map(p => {

          if (m == 0) {
            return
          }
          const ms = m.toString()
          let newPat = p

          if (ms.includes('R')) {
            newPat = newPat.restart(1)
          }
          if (ms.includes('B')) {
            newPat = newPat.rev().speed(-1)
          }
          modifiers.forEach(([mod, callback]) => {
            if (mod(ms)) {
              newPat = callback(newPat)
            }
          })
          return newPat
        }).filter(Boolean))
      }).innerJoin()
    }).flat()
  )
}

サクノバ

Ableton Live と Strudel を使って、電子音楽を作っています。 テクノ、ハイパーポップ、トランスなどが好きで、最近は特に Strudel でコードを書きながら 音楽を組み立てています。

Strudel に惹かれた一番の理由は、音楽ができあがっていく「過程」そのものを 演奏にできることです。

キックを鳴らして、ベースを足して、コードを書いて、音を崩して、また組み直す。 何もないところから少しずつ音楽ができていく。その一連の流れまで含めてパフォーマンスになる ところが、ライブコーディングの面白さだと思っています。

音楽を完成させること自体は、以前よりずっと簡単になりました。 だからこそ、自分は完成した曲だけではなく、「どうやってそこに辿り着いたのか」まで 見せられる音楽をやってみたいと思っています。

今は Strudel で作った作品や、好きな曲のコピーなどを投稿しながら試行錯誤しています。 TikTok では、ときどき Strudel を使ったライブ配信もしています。

そして今取り組んでいるのが、Ableton Live と Strudel をリアルタイムに 同期させた演奏です。

Strudel でコードを書きながらパターンを変化させ、Ableton の音やエフェクトと組み合わせて、 その場で音楽を組み立てていく。最終的には、PC の前だけではなく、人前に立って Strudel でライブをすることがひとつの目標です。

このサイトを作ったのも、その活動の延長です。

Strudel はとても面白いツールなのですが、日本語で調べようとすると、まだ情報がかなり 限られています。自分自身、分からないことをひとつずつ調べながら使ってきました。

だったら、自分が覚えたことや作ったものを残しておけば、これから Strudel を触る人の 役に立つかもしれない。

そんな理由から、このサイトにはStrudel の関数をまとめた早見表や、 そのまま部品として使ったり改造したりできるコード・作品を置いています。

まだ自分も試している途中です。

このサイトや作品をきっかけに Strudel を触る人が少しでも増えて、 「こうしたらもっと面白い」「こんな使い方もできる」と、いろんな人と試行錯誤できたら嬉しいです。
