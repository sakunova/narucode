// Sakanaction - Music

const master = slider(0.667, 0, 1)
all(p => p.mul(gain(master)))


const _rlpfHz = (knob, knobMax=5, maxHz=5000) =>
  knob.fmap(x => (x / knobMax) * maxHz)

Pattern.prototype.rlpf = function(knob){
  return this.lpf(_rlpfHz(knob, 5, 5000))
}

window.ar = function(...input) {

  if (input.length % 2 !== 0) {
    throw new Error('Arrange needs a length paramter for each pattern (length, pattern, length, pattern)');
  }
  let sects = [];
  let total = 0;
  for (let i = 0; i < input.length; i += 2) {
    let inp = [input.at(i)].flat()
    let cycles = inp.at(0);
    let start = inp.at(1) ?? 0
    total += cycles;

    let pattern = input.at(i + 1);
    sects.push([cycles, pattern.ribbon(start, cycles).fast(cycles)]);
  }
  return stepcat(...sects).slow(total);
}


window.p = stack

window.blockArrange = function (patArr, modifiers = []) {
  return stack(
    ...patArr.map(([pat, maskPat]) => {
      pat = [pat].flat()

      return maskPat.fmap(m => {
        return stack(...pat.map(p => {

          if (m == 0) {
            return
          }
          const ms = m.toString()
          let newPat = p

          if (ms.includes('R')) {
            newPat = newPat.restart(1)
          }
          if (ms.includes('B')) {
            newPat = newPat.rev().speed(-1)
          }
          modifiers.forEach(([mod, callback]) => {
            if (mod(ms)) {
              newPat = callback(newPat)
            }
          })
          return newPat
        }).filter(Boolean))
      }).innerJoin()
    }).flat()
  )
}

sah-koo-NOH-vah

I make electronic music with Ableton Live and Strudel. I'm into techno, hyperpop and trance, and lately I build most of my tracks by writing code in Strudel.

What pulled me into Strudel is that the process of a track coming together can itself be the performance.

Start a kick, add a bass, write a chord, break it, put it back together. Music appearing out of nothing, one line at a time. That the whole path is part of the show is what makes live coding interesting to me.

Finishing a track has never been easier than it is now. Which is exactly why I want to make music that shows how you got there, not just the finished thing.

Right now I post tracks I made in Strudel and covers of songs I like, figuring it out as I go. I sometimes livestream Strudel sessions on TikTok.

The thing I'm working on now is playing Ableton Live and Strudel in real-time sync.

Morphing patterns by writing code in Strudel, running it against Ableton's sounds and effects, building the track on the spot. Eventually I want to do that in front of an audience, not just in front of my laptop.

This site came out of that.

Strudel's own docs are good, but there's no single searchable list of every function — and some of the example code that ships with it doesn't actually make a sound in a browser. It quietly assumes SuperCollider, a separate program you'd have to install. I found that out by pasting an example and getting silence.

So the reference here is cut down to functions that run in the browser on their own, and the examples I wrote myself are executed at build time — if one of them doesn't make a sound, the build fails and the page never ships.

That's what this site is: a reference for Strudel's functions and real track code you can copy, take apart, and rebuild.

I'm still figuring it out myself.

If this gets a few more people writing Strudel — and someone tells me "this would be better if you did it that way" — that's the whole point.
